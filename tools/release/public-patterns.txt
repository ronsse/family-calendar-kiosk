# Generic secret and environment shapes that must never appear in the public tree.
# One POSIX extended regular expression per line, matched case-sensitively.
# verify-clean.sh ignores lines that start with '#' and blank lines.
# A line of the form 'allow-re <ERE>' (no '/' in it) names a placeholder that is
# removed from each candidate line, wherever it is not followed by another digit,
# before the patterns are applied again. Written as an escaped ERE rather than a
# literal so that this file does not itself trip the household term grep. Use it only
# for documented placeholders, never to silence a real value.
#
# Google Calendar "secret address in iCal format": .../private-<hex>/basic.ics
private-[0-9a-f]{16,}
# JWT shape (Home Assistant long-lived access tokens)
eyJ[A-Za-z0-9_-]{10,}\.eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}
# NVIDIA API keys
nvapi-[A-Za-z0-9_-]{16,}
# Tailscale auth and API keys
tskey-[A-Za-z0-9_-]{16,}
# 1Password service-account tokens
ops_[A-Za-z0-9_-]{20,}
# fal.ai keys: <uuid>:<32 hex>
[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}:[0-9a-f]{32}
# Tailscale CGNAT range 100.64.0.0/10: an address in it is a real tailnet IP
(^|[^0-9.])100\.(6[4-9]|[7-9][0-9]|1[01][0-9]|12[0-7])\.[0-9]{1,3}\.[0-9]{1,3}($|[^0-9])
# Host data mounts and home directories
/mnt/
/home/[a-z]+
#
# Allowed placeholders.
# Tailscale's MagicDNS resolver is the same fixed address on every tailnet.
allow-re 100\.100\.100\.100
# The approved documentation placeholders for the server and the Pi.
allow-re 100\.64\.0\.2
allow-re 100\.64\.0\.3
# The whole Tailscale CGNAT range, written as a CIDR in ufw and trusted_networks examples.
allow-re 100\.64\.0\.0.10
